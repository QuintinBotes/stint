// Held out: no agent sees this file. The daemon copies it into a checkout of the task branch after the run.
// Written from the task's goal and acceptance only: the event may come back bare or as { event }.
import { describe, expect, it } from 'vitest';
import { createApp, createDeps } from '../src/app';
import { loadConfig } from '../src/config';

function setup() {
  const app = createApp(createDeps(loadConfig({})));
  const send = (path: string, tenant?: string, init: RequestInit = {}) => {
    const headers = new Headers(init.headers);
    if (tenant) headers.set('X-Tenant', tenant);
    if (init.body) headers.set('Content-Type', 'application/json');
    return app.request(path, { ...init, headers });
  };
  const created = async (tenant: string, type: string) =>
    (await (await send('/webhooks', tenant, { method: 'POST', body: JSON.stringify({ type, data: { n: 1 } }) })).json()) as { id: string };
  return { send, created };
}

describe('held out: GET /events/:id', () => {
  it('returns the stored event to the tenant that owns it', async () => {
    const { send, created } = setup();
    const { id } = await created('tenant-a', 'a.one');
    const res = await send(`/events/${id}`, 'tenant-a');
    expect(res.status).toBe(200);
    const body = (await res.json()) as { event?: unknown };
    expect(body.event ?? body).toMatchObject({ id, tenant: 'tenant-a', type: 'a.one' });
  });

  it('answers 404 when another tenant asks for the event', async () => {
    const { send, created } = setup();
    const { id } = await created('tenant-a', 'a.one');
    expect((await send(`/events/${id}`, 'tenant-b')).status).toBe(404);
  });

  it('answers 404 for an event that does not exist', async () => {
    const { send } = setup();
    expect((await send('/events/evt_999999', 'tenant-a')).status).toBe(404);
  });

  it('answers 400 without the X-Tenant header', async () => {
    const { send, created } = setup();
    const { id } = await created('tenant-a', 'a.one');
    expect((await send(`/events/${id}`)).status).toBe(400);
  });
});
